using UnityEngine; using UnityEngine.Video; [RequireComponent(typeof(VideoPlayer))] public class ARVideoPlayer : MonoBehaviour { [Header("Налаштування відео")] [Tooltip("Посилання на відео")] public string videoUrl = "https://files.catbox.moe/gswdrf.mov"; public bool loopVideo = true; private VideoPlayer videoPlayer; void Awake() { videoPlayer = GetComponent(); // Базові налаштування videoPlayer.source = VideoSource.Url; videoPlayer.url = videoUrl; videoPlayer.playOnAwake = false; videoPlayer.isLooping = loopVideo; // Рендеринг на RenderTexture або Material (залежить від твоїх налаштувань) // videoPlayer.renderMode = VideoRenderMode.MaterialOverride; // Починаємо підготовку відео у фоні videoPlayer.Prepare(); } // Викликається системою AR (Vuforia / AR Foundation), коли маркер знайдено public void OnTrackingFound() { Debug.Log("Маркер знайдено. Запуск відео..."); if (videoPlayer.isPrepared) { videoPlayer.Play(); } else { // Якщо відео ще не завантажилось з мережі, чекаємо і запускаємо одразу як буде готово videoPlayer.prepareCompleted += PlayWhenPrepared; } } // Викликається, коли маркер втрачено public void OnTrackingLost() { Debug.Log("Маркер втрачено. Пауза відео..."); // Відписуємось від події, щоб уникнути багів videoPlayer.prepareCompleted -= PlayWhenPrepared; if (videoPlayer.isPlaying) { videoPlayer.Pause(); // Якщо хочеш, щоб відео щоразу починалось з початку, заміни Pause() на Stop() } } private void PlayWhenPrepared(VideoPlayer vp) { vp.Play(); vp.prepareCompleted -= PlayWhenPrepared; } }
